    function sendOTP() {
        let phone = document.getElementById("userPhone").value;
        if(phone.length === 10) {
            document.getElementById("otpBox").style.display = "block";
            alert("डिफ़ॉल्ट OTP (1234) भेजा गया है।");
        } else {
            alert("कृपया सही 10 अंकों का मोबाइल नंबर दर्ज करें।");
        }
    }

    function verifyOTP() {
        let otp = document.getElementById("enteredOtp").value;
        if(otp === "1234") {
            alert("सफलतापूर्वक लॉगिन हो गया!");
            document.getElementById("loginScreen").style.display = "none";
            document.getElementById("mainAppScreen").style.display = "block";
            displayListings();
        } else {
            alert("गलत OTP! कृपया '1234' दर्ज करें।");
        }
    }

    function switchPanel(panelName) {
        document.querySelectorAll('.panel-section').forEach(p => p.classList.remove('active'));
        document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));

        if(panelName === 'student') {
            document.getElementById('studentPanel').classList.add('active');
            event.target.classList.add('active');
            displayListings();
        } else if(panelName === 'owner') {
            document.getElementById('ownerPanel').classList.add('active');
            event.target.classList.add('active');
        } else if(panelName === 'admin') {
            document.getElementById('adminPanel').classList.add('active');
            event.target.classList.add('active');
        }
    }

    function displayListings(filterArea = "All") {
        let container = document.getElementById("listingContainer");
        container.innerHTML = "";
        let filtered = listings.filter(item => filterArea === "All" || item.mainArea === filterArea);

        filtered.forEach(item => {
            let mapLink = `https://www.google.com/maps/search/?api=1&query=${encodeURIComponent(item.subLocation + ", Jamshedpur")}`;
            container.innerHTML += `
                <div class="card">
                    <span class="tag">${item.type} | 📍 ${item.mainArea}</span>
                    <h3>${item.name}</h3>
                    <p style="margin:3px 0; font-size:13px; color:#64748b;"><b>लोकेशन:</b> ${item.subLocation}</p>
                    <p class="price">💰 ${item.price}</p>
                    <p style="margin:3px 0; font-size:13px;">🛠️ ${item.features}</p>
                    <div class="btn-group">
                        <a href="https://wa.me/${item.contact}?text=नमस्ते, मैंने ₹99 पास पे किया है। मुझे ${item.name} में कमरा चाहिए।" class="whatsapp-btn" target="_blank">💬 ओनर से संपर्क (₹99 Pass)</a>
                        <a href="${mapLink}" class="map-btn" target="_blank">🗺️ मैप लोकेशन</a>
                    </div>
                </div>
            `;
        });
    }

    function addListing() {
        let type = document.getElementById("listType").value;
        let ownerName = document.getElementById("ownerName").value;
        let name = document.getElementById("name").value;
        let mainArea = document.getElementById("mainArea").value;
        let subLocation = document.getElementById("subLocation").value;
        let price = document.getElementById("priceDetails").value;
        let features = document.getElementById("features").value;
        let contact = document.getElementById("contact").value;

        if(ownerName && name && subLocation && price && contact) {
            listings.push({ type, name: `${name} (ओनर: ${ownerName})`, mainArea, subLocation, price, features, contact });
            document.getElementById("listingForm").reset();
            alert("KYC वेरिफाइड! प्रॉपर्टी सफलतापूर्वक लाइव हो गई है।");
            switchPanel('student');
        } else {
            alert("कृपया सभी जानकारी भरें।");
        }
    }

    function filterListings() {
        let selectedArea = document.getElementById("areaFilter").value;
        displayListings(selectedArea);
    }
